# Certificates for cert-inventory to monitor. One per line:
#
#   namespace   kind        name             key
#
# kind  secret or configmap
# key   the data key holding the certificate (PEM, PEM bundle, or DER),
#       or * to read every key in the object that contains a PEM certificate
#       (keys without a certificate, such as tls.key, are ignored)
#
# After editing this file, regenerate the RBAC so the exporter may read exactly
# these objects (and nothing else):
#
#   python deploy/cert_inventory.py --print-rbac deploy/targets.txt > deploy/rbac-targets.yaml
#
# The entry below exists on every cluster (the cluster CA that Kubernetes
# publishes into each namespace), so a fresh deploy shows data immediately.
# Replace or extend it with your own secrets, for example:
#   my-app        secret      my-app-tls        tls.crt
#   my-app        secret      my-app-tls        ca.crt
#   ingress       secret      wildcard-cert     *
#   kube-system   secret      rke2-serving      tls.crt

kube-system     configmap   kube-root-ca.crt  ca.crt
